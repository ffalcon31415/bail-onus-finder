""" Bail Onus Finder - Streamlit app. A questionnaire that determines whether bail is CROWN ONUS or REVERSE ONUS. If reverse onus, the triggering section numbers (only) are shown in a copyable box (click the copy icon in its top-right corner). Reads the same editable rule file as the AutoHotkey version: OnusRules.txt (kept in the same folder as this script). Edit that file to change the questionnaire - no code changes needed. Run with: streamlit run onus_app.py """ import streamlit as st def load_rules(): """Parse OnusRules.txt into two ordered lists of (description, section). Mirrors the AutoHotkey loader: [ALWAYS] -> group A (apply regardless of classification) [INDICTABLE] -> group B (indictable / hybrid-not-summary track only) "section | description" per line; ';' lines are comments. """ group_a, group_b = [], [] target = None rules_str = """[ALWAYS] 515(6)(c) | Offence under s.145(2)-(5) while at large after being released for another offence under CC 515(6)(b.1)(i) | IPV (incl. sexual offence, crim. harassment, HT) AND accused was previously convicted or discharged for IPV offence (discharge within retention period) 515(6)(b.1)(ii) | IPV and at the time of the offence accused was on an IPV peace bond under CC 515(6)(d) | CDSA offence punishable by life under ss.5-7, or a conspiracy for such an offence 742.6(2) | Breach of a Conditional Sentence Order 524(4)(a) | 524 application (once granted) if the justice finds the accused has contravened or had been about to contravene a release term 524(4)(c) | 524 application (once granted) if any of the outstanding charges is reverse onus due to 515(6) or is a 469 offence 523(2.1) | Following a finding of guilt, Crown applied to vacate the release order 522(2) | 469 Offence [INDICTABLE] 524(4)(b) | 524 application (once granted) 515(6)(a)(i) | On release for an indictable or hybrid-not-elected-summarily offence 515(6)(a)(ix) | Any degree of assault or sexual assault where choking, suffocating or strangling is alleged 515(6)(b) | Not ordinarily resident in Canada 515(6)(a)(viii) | Firearm / ammo / prohibited weapon etc. while on a prohibition order barring possession of same 515(6)(a)(vi) | Certain firearm offences (ss.95, 98, 98.1, 99, 100, 102 or 103) 515(6)(a)(vi) to (vii) | Certain offences committed with a firearm (ss.244, 244.2, 239, 272, 273, 279(1), 279.1, 344 or 346) 515(6)(a)(x) | Human trafficking under s.279.01 or 279.011 515(6)(a)(xi) | Motor vehicle theft with violence under s.333.1(3) or for the benefit of a criminal org under s.333.1(4) 515(6)(a)(xii) | Extortion with violence used, threatened, or attempted 515(6)(a)(xii.1) | Violent offence AND two or more prior violent offences prosecuted by indictment that resulted in conviction or absolute or (xiv) """ for raw in rules_str.splitlines(): line = raw.strip() if not line or line.startswith(";"): continue if line.startswith("[") and line.endswith("]"): name = line[1:-1].strip().upper() target = {"ALWAYS": "A", "INDICTABLE": "B"}.get(name) continue if "|" not in line or target is None: continue section, _, desc = line.partition("|") section, desc = section.strip(), desc.strip() if not section or not desc: continue (group_a if target == "A" else group_b).append((desc, section)) return group_a, group_b def dedupe(seq): """Return items in order, dropping later duplicates.""" seen, out = set(), [] for item in seq: if item not in seen: seen.add(item) out.append(item) return out st.set_page_config(page_title="Bail Onus Finder", page_icon="⚖️") st.title("⚖️ Bail Onus Finder") st.caption("Reflects amendments in effect July 18, 2026") st.caption("Below may contain errors. You are ultimately responsible for ensuring the onus is correctly noted as per Criminal Code.") group_a, group_b = load_rules() if not group_a and not group_b: st.error("No rules were loaded. Check that OnusRules.txt contains " "[ALWAYS] / [INDICTABLE] sections with 'section | description' lines.") st.stop() # --- Step 1: Youth vs Adult -------------------------------------------------- matter = st.radio( "Is this a youth matter or an adult matter?", ["Adult", "Youth (YCJA)"], index=0, ) if matter.startswith("Youth"): st.divider() st.subheader("Result: CROWN ONUS") st.write("Onus for youth matters is always Crown Onus (YCJA).") st.stop() # --- Step 2: Checklist A (applies regardless of classification) -------------- st.divider() st.subheader("Do any of these apply?") st.caption("These trigger reverse onus regardless of offence classification.") picked = [] for i, (desc, section) in enumerate(group_a): if st.checkbox(desc, key=f"a{i}"): picked.append((desc, section)) # --- Step 3: Indictable gate ------------------------------------------------- st.divider() st.subheader("Offence classification") indictable = st.radio( "Do the charges for which the accused is applying for bail include a straight indictable offence, OR a hybrid offence for " "which the Crown has NOT elected to proceed summarily?", ["No", "Yes"], index=0, ) # --- Step 4: Checklist B (only on the indictable / not-summary track) -------- if indictable == "Yes": st.divider() st.subheader("Do any of these apply?") st.caption("Indictable or hybrid offence charged.") for i, (desc, section) in enumerate(group_b): if st.checkbox(desc, key=f"b{i}"): picked.append((desc, section)) # --- Result ------------------------------------------------------------------ st.divider() if not picked: st.subheader("Result: CROWN ONUS") st.write("None of the reverse-onus conditions apply.") else: sections = dedupe(section for _desc, section in picked) st.subheader("Result: REVERSE ONUS") st.write("Onus on the accused to show cause why \"detention in custody is not justified by _clearly demonstrating_ that their proposed release plan addresses the risks posed by the accused\".") st.write("**Section numbers** (click the copy icon to copy):") st.code(", ".join(sections), language=None) st.write("**Reasons:**") for desc, section in picked: st.markdown(f"- {desc} → **{section}**")
